0 c10 "//"

0 c10 "// NaN and +/-Inf values will return an [UnsupportedValueError]."

0 c10 "// This replacement can be disabled when using an [Encoder],"
0 c10 "// by calling [Encoder.SetEscapeHTML](false)."
0 c10 "//"
0 c10 "// Array and slice values encode as JSON arrays, except that"
0 c10 "// []byte encodes as a base64-encoded string, and a nil slice"

0 c10 "//"

0 c10 "//\tField int `json:\",omitempty\"`"
0 c10 "//"

0 c10 "//\tField int `json:\"-\"`"
