getu4(s)rune{
if len(s)<6||s[0]!='\\'||s[1]!='u'{return-1}
var r rune for c in s[2:6]{

case '0'<=c&&c<='9':
c=c-'0'
case 'a'<=c&&c<='f':
c=c-'a'+10
case 'A'<=c&&c<='F':
c=c-'A'+10
default:
return-1
}
r=r*16+rune(c)
}
return r
}

case '\r':

case '\t':

i++
start=i continue

dst=append(dst '\\' 'u' '2' '0' '2' hex[c&0xF])
i+=size
start=i continue
}
i+=size
}
dst=append(dst src[start:]...)
dst=append(dst '"')
return dst
}

case Bool,

Uint Uint8 Uint16 Uint32 Uint64 Uintptr,

}
}

}

v.Interface().(isZeroer).IsZero()
}

}
case t.Implements(isZeroerType):

}

appendIndent(\dst,src[]byte,prefix,indent string)[]byte!{

scan:=newScanner()
defer freeScanner(scan)
needIndent:=false
depth:=0 for c in src{
scan.bytes++
v:=scan.step(scan c)
if v==scanSkipSpace{continue}
if v==scanError{break}
if needIndent&&v!=scanEndObject&&v!=scanEndArray{
needIndent=false
depth++
dst=appendNewline(dst prefix indent depth)
}
if v==scanContinue{
dst=append(dst c)
continue
}
switch c{
case '{' '[':
needIndent=true
dst=append(dst c)
case ',':
dst=append(dst c)
dst=appendNewline(dst prefix indent depth)
case ':':
dst=append(dst c ' ')
case '}' ']':
if needIndent{
needIndent=false
}else{
depth--
dst=appendNewline(dst prefix indent depth)
}
dst=append(dst c)
default:
dst=append(dst c)
}
}

^dst

step func(*scanner,byte)int
endTop bool
parseState[]int
err error
bytes int64
}
var scannerPool=sync.Pool{
New:func()any{
return&scanner{}
},
}
newScanner()*scanner{
scan:=scannerPool.Get().(*scanner)
scan.bytes=0
scan.reset()
return scan
}
freeScanner(scan*scanner){
if len(parseState)>1024{parseState=nil}
scannerPool.Put(scan)
}
const(
scanContinue=iota
scanBeginLiteral
scanBeginObject
scanObjectKey
scanObjectValue
scanEndObject
scanBeginArray
scanArrayValue
scanEndArray
scanSkipSpace
scanEnd
scanError
)
const(
parseObjectKey=iota
parseObjectValue
parseArrayValue
)
const maxNestingDepth=10000
(s)reset(){
step=stateBeginValue
parseState=parseState[0:0]
err=nil
endTop=false
}
.eof()int{
if err!=nil{return scanError}
if endTop{return scanEnd}
step(s ' ')
if endTop{return scanEnd}
if err==nil{err=&SyntaxError{"unexpected end of JSON input",s.bytes}}
return scanError
}
.pushParseState(c,newParseState int,successState int)int{
parseState=append(parseState newParseState)
if len(parseState)<=maxNestingDepth{return successState}
return s.error(c "exceeded max depth")
}
.popParseState(){
n:=len(parseState)-1
parseState=parseState[0:n]
if n==0{
step=stateEndTop
endTop=true
}else{step=stateEndValue}
}
isSpace(c)bool=>c<=' '&&(c==' '||c=='\t'||c=='\r'||c=='\n')
stateBeginValueOrEmpty(s,c)int{
if isSpace(c){return scanSkipSpace}
if c==']'{return stateEndValue(s c)}
return stateBeginValue(s c)
}
stateBeginValue(s,c)int{
if isSpace(c){return scanSkipSpace}
switch c{
case '{':
step=stateBeginStringOrEmpty return pushParseState(c parseObjectKey scanBeginObject)
case '[':
step=stateBeginValueOrEmpty return pushParseState(c parseArrayValue scanBeginArray)
case '"':
step=stateInString return scanBeginLiteral
case '-':
step=stateNeg return scanBeginLiteral
case '0':
step=state0 return scanBeginLiteral
case 't':
step=stateT return scanBeginLiteral
case 'f':
step=stateF return scanBeginLiteral
case 'n':
step=stateN return scanBeginLiteral
}
if '1'<=c&&c<='9'{
step=state1 return scanBeginLiteral
}
return s.error(c "looking for beginning of value")
}
stateBeginStringOrEmpty(s,c)int{
if isSpace(c){return scanSkipSpace}
if c=='}'{
n:=len(parseState)
parseState[n-1]=parseObjectValue return stateEndValue(s c)
}
return stateBeginString(s c)
}
stateBeginString(s,c)int{
if isSpace(c){return scanSkipSpace}
if c=='"'{
step=stateInString return scanBeginLiteral
}
return s.error(c "looking for beginning of object key string")
}
stateEndValue(s,c)int{
n:=len(parseState)
if n==0{
step=stateEndTop
endTop=true return stateEndTop(s c)
}
if isSpace(c){
step=stateEndValue return scanSkipSpace
}
ps:=parseState[n-1]
switch ps{
case parseObjectKey:
if c==':'{
parseState[n-1]=parseObjectValue
step=stateBeginValue return scanObjectKey
}
return s.error(c "after object key")
case parseObjectValue:
if c==','{
parseState[n-1]=parseObjectKey
step=stateBeginString return scanObjectValue
}
if c=='}'{
popParseState()
return scanEndObject
}
return s.error(c "after object key:value pair")
case parseArrayValue:
if c==','{
step=stateBeginValue return scanArrayValue
}
if c==']'{
popParseState()
return scanEndArray
}
return s.error(c "after array element")
}
return s.error(c "")
}
stateEndTop(s,c)int{
if!isSpace(c){
s.error(c "after top-level value")
}
return scanEnd
}
stateInString(s,c)int{
if c=='"'{
step=stateEndValue return scanContinue
}
if c=='\\'{
step=stateInStringEsc return scanContinue
}
if c<0x20{return s.error(c "in string literal")}
return scanContinue
}
stateInStringEsc(s,c)int{
switch c{
case 'b' 'f' 'n' 'r' 't' '\\' '/' '"':
step=stateInString return scanContinue
case 'u':
step=stateInStringEscU return scanContinue
}
return s.error(c "in string escape code")
}
stateInStringEscU(s,c)int{
if '0'<=c&&c<='9'||'a'<=c&&c<='f'||'A'<=c&&c<='F'{
step=stateInStringEscU1 return scanContinue
}
return s.error(c "in \\u hexadecimal character escape")
}
stateInStringEscU1(s,c)int{
if '0'<=c&&c<='9'||'a'<=c&&c<='f'||'A'<=c&&c<='F'{
step=stateInStringEscU12 return scanContinue
}
return s.error(c "in \\u hexadecimal character escape")
}
stateInStringEscU12(s,c)int{
if '0'<=c&&c<='9'||'a'<=c&&c<='f'||'A'<=c&&c<='F'{
step=stateInStringEscU123 return scanContinue
}
return s.error(c "in \\u hexadecimal character escape")
}
stateInStringEscU123(s,c)int{
if '0'<=c&&c<='9'||'a'<=c&&c<='f'||'A'<=c&&c<='F'{
step=stateInString return scanContinue
}
return s.error(c "in \\u hexadecimal character escape")
}
stateNeg(s,c)int{
if c=='0'{
step=state0 return scanContinue
}
if '1'<=c&&c<='9'{
step=state1 return scanContinue
}
return s.error(c "in numeric literal")
}
state1(s,c)int{
if '0'<=c&&c<='9'{
step=state1 return scanContinue
}
return state0(s c)
}
state0(s,c)int{
if c=='.'{
step=stateDot return scanContinue
}
if c=='e'||c=='E'{
step=stateE return scanContinue
}
return stateEndValue(s c)
}
stateDot(s,c)int{
if '0'<=c&&c<='9'{
step=stateDot0 return scanContinue
}
return s.error(c "after decimal point in numeric literal")
}
stateDot0(s,c)int{
if '0'<=c&&c<='9'{return scanContinue}
if c=='e'||c=='E'{
step=stateE return scanContinue
}
return stateEndValue(s c)
}
stateE(s,c)int{
if c=='+'||c=='-'{
step=stateESign return scanContinue
}
return stateESign(s c)
}
stateESign(s,c)int{
if '0'<=c&&c<='9'{
step=stateE0 return scanContinue
}
return s.error(c "in exponent of numeric literal")
}
stateE0(s,c)int{
if '0'<=c&&c<='9'{return scanContinue}
return stateEndValue(s c)
}
stateT(s,c)int{
if c=='r'{
step=stateTr return scanContinue
}
return s.error(c "in literal true (expecting 'r')")
}
stateTr(s,c)int{
if c=='u'{
step=stateTru return scanContinue
}
return s.error(c "in literal true (expecting 'u')")
}
stateTru(s,c)int{
if c=='e'{
step=stateEndValue return scanContinue
}
return s.error(c "in literal true (expecting 'e')")
}
stateF(s,c)int{
if c=='a'{
step=stateFa return scanContinue
}
return s.error(c "in literal false (expecting 'a')")
}
stateFa(s,c)int{
if c=='l'{
step=stateFal return scanContinue
}
return s.error(c "in literal false (expecting 'l')")
}
stateFal(s,c)int{
if c=='s'{
step=stateFals return scanContinue
}
return s.error(c "in literal false (expecting 's')")
}
stateFals(s,c)int{
if c=='e'{
step=stateEndValue return scanContinue
}
return s.error(c "in literal false (expecting 'e')")
}
stateN(s,c)int{
if c=='u'{
step=stateNu return scanContinue
}
return s.error(c "in literal null (expecting 'u')")
}
stateNu(s,c)int{
if c=='l'{
step=stateNul return scanContinue
}
return s.error(c "in literal null (expecting 'l')")
}
stateNul(s,c)int{
if c=='l'{
step=stateEndValue return scanContinue
}
return s.error(c "in literal null (expecting 'l')")
}
stateError(s,c)int=>scanError
.error(c,context string)int{
step=stateError
err=&SyntaxError{"invalid character "+quoteChar(c)+" "+context,s.bytes}
return scanError
}
quoteChar(c)string{
if c=='\''{return `'\''`}
if c=='"'{return `'"'`}
s:=strconv.Quote(string(c))
return "'"+s[1:len(s)-1]+"'"
}

}
NewDecoder(r Reader)*Decoder=>&Decoder{r:r}
(dec)UseNumber(){d.useNumber=true}
.DisallowUnknownFields(){d.disallowUnknownFields=true}
.Decode(v any)!{
if err!=nil{return err}
tokenPrepareForDecode()?
if!tokenValueAllowed(){return&SyntaxError{msg:"not at beginning of value",Offset:InputOffset()}}
n:=readValue()?
d.init(buf[scanp:scanp+n])
scanp+=n
err=d.unmarshal(v)
tokenValueEnd()
return err
}
.Buffered()Reader=>bytes.NewReader(buf[scanp:])
.readValue()int!{
scan.reset()
scanp:=scanp var err error
Input:
for scanp>=0{
for;scanp<len(buf);scanp++{
c:=buf[scanp]
scan.bytes++
switch scan.step(&scan c){
case scanEnd:
scan.bytes--
break Input
case scanEndObject scanEndArray:
if stateEndValue(&scan ' ')==scanEnd{
scanp++
break Input
}
case scanError:
dec.err=scan.err
!scan.err
}
}
if err!=nil{
if err==EOF{
if scan.step(&scan ' ')==scanEnd{break Input}
if nonSpace(buf){err=ErrUnexpectedEOF}
}
dec.err=err
!err
}
n:=scanp-dec.scanp
err=refill()
scanp=dec.scanp+n
}
^scanp-dec.scanp
}
.refill()!{
if scanp>0{
scanned+=int64(scanp)
n:=copy(buf buf[scanp:])
buf=buf[:n]
scanp=0
}
const minRead=512 if cap(buf)-len(buf)<minRead{
newBuf:=make([]byte len(buf),2*cap(buf)+minRead)
copy(newBuf buf)
buf=newBuf
}
n,err:=r.Read(buf[len(buf):cap(buf)])
buf=buf[0:len(buf)+n]
return err
}

type Token any
const(
tokenTopValue=iota
tokenArrayStart
tokenArrayValue
tokenArrayComma
tokenObjectStart
tokenObjectKey
tokenObjectColon
tokenObjectValue
tokenObjectComma
)
(dec)tokenPrepareForDecode()!{
switch tokenState{
case tokenArrayComma:
c,err:=peek()
if err!=nil{return err}
if c!=','{return&SyntaxError{"expected comma after array element",InputOffset()}}
scanp++
tokenState=tokenArrayValue
case tokenObjectColon:
c,err:=peek()
if err!=nil{return err}
if c!=':'{return&SyntaxError{"expected colon after object key",InputOffset()}}
scanp++
tokenState=tokenObjectValue
}
^
}
.tokenValueAllowed()bool{
switch tokenState{
case tokenTopValue tokenArrayStart tokenArrayValue tokenObjectValue:
return true
}
return false
}
.tokenValueEnd(){
switch tokenState{
case tokenArrayStart tokenArrayValue:
tokenState=tokenArrayComma
case tokenObjectValue:
tokenState=tokenObjectComma
}
}
type Delim rune
(d Delim)String()string=>string(d)
(dec)Token()Token!{
for{
c:=peek()?
switch c{
case '[':
if!tokenValueAllowed(){return tokenError(c)}
scanp++
tokenStack=append(tokenStack tokenState)
tokenState=tokenArrayStart
^Delim('[')
case ']':
if tokenState!=tokenArrayStart&&tokenState!=tokenArrayComma{return tokenError(c)}
scanp++
tokenState=tokenStack[len(tokenStack)-1]
tokenStack=tokenStack[:len(tokenStack)-1]
tokenValueEnd()
^Delim(']')
case '{':
if!tokenValueAllowed(){return tokenError(c)}
scanp++
tokenStack=append(tokenStack tokenState)
tokenState=tokenObjectStart
^Delim('{')
case '}':
if tokenState!=tokenObjectStart&&tokenState!=tokenObjectComma{return tokenError(c)}
scanp++
tokenState=tokenStack[len(tokenStack)-1]
tokenStack=tokenStack[:len(tokenStack)-1]
tokenValueEnd()
^Delim('}')
case ':':
if tokenState!=tokenObjectColon{return tokenError(c)}
scanp++
tokenState=tokenObjectValue continue
case ',':
if tokenState==tokenArrayComma{
scanp++
tokenState=tokenArrayValue continue
}
if tokenState==tokenObjectComma{
scanp++
tokenState=tokenObjectKey continue
}
return tokenError(c)
case '"':
if tokenState==tokenObjectStart||tokenState==tokenObjectKey{
var x string
old:=tokenState
tokenState=tokenTopValue
err:=Decode(&x)
tokenState=old if err!=nil{!err}
tokenState=tokenObjectColon
^x
}
fallthrough
default:
if!tokenValueAllowed(){return tokenError(c)}
var x any if err:=Decode(&x);err!=nil{!err}
^x
}
}
}
.tokenError(c byte)Token!{
var context string switch tokenState{
case tokenTopValue:
context=" looking for beginning of value"
case tokenArrayStart tokenArrayValue tokenObjectValue:
context=" looking for beginning of value"
case tokenArrayComma:
context=" after array element"
case tokenObjectKey:
context=" looking for beginning of object key string"
case tokenObjectColon:
context=" after object key"
case tokenObjectComma:
context=" after object key:value pair"
}
!&SyntaxError{"invalid character "+quoteChar(c)+context,InputOffset()}
}
.More()bool{
c,err:=peek()
return err==nil&&c!=']'&&c!='}'
}
.peek()byte!{
var err error for{
for i:=scanp;i<len(buf);i++{
c:=buf[i]
if isSpace(c){continue}
scanp=i
^c
}
if err!=nil{!err}
err=refill()
}
}
.InputOffset()int64=>scanned+int64(scanp)
